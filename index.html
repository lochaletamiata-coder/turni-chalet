// ═══════════════════════════════════════════════════════════════ // APPS SCRIPT - Lo Chalet Amiata · Turni Staff // ═══════════════════════════════════════════════════════════════ const SHEET_NAME = 'Turni'; function doGet(e) { const action = e.parameter.action; const callback = e.parameter.callback; // JSONP callback let result; try { if (action === 'load') { result = handleLoad(e.parameter.keys); } else if (action === 'save') { const data = JSON.parse(e.parameter.data); handleSave(data); result = {ok: true}; } else { result = {ok: true}; } } catch(err) { result = {error: err.message}; } const json = JSON.stringify(result); // Se c'è callback → risposta JSONP, altrimenti JSON normale if (callback) { return ContentService.createTextOutput(`${callback}(${json})`) .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(SHEET_NAME); if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.appendRow(['data','fascia','riga','nome','entrata','uscita','mansione','annotazioni']); sh.getRange(1,1,1,8).setFontWeight('bold').setBackground('#3B2A1A').setFontColor('#F5E6C8'); sh.setFrozenRows(1); sh.setColumnWidth(1,100);sh.setColumnWidth(2,180);sh.setColumnWidth(3,50); sh.setColumnWidth(4,120);sh.setColumnWidth(5,80);sh.setColumnWidth(6,80); sh.setColumnWidth(7,100);sh.setColumnWidth(8,220); } return sh; } function handleLoad(keysParam) { const keys = keysParam ? keysParam.split(',') : []; const sh = getSheet(); const data = sh.getDataRange().getValues(); const result = {}; keys.forEach(k => { result[k] = { fasce: {}, annotazioni: '' }; ['APERTURA','PRANZO / SELF SERVICE','POMERIGGIO','CENA'].forEach(f => { const maxRows = f === 'APERTURA' ? 4 : 7; result[k].fasce[f] = Array.from({length: maxRows}, () => ({nome:'', entrata:'', uscita:'', mansione:''})); }); }); for (let i = 1; i < data.length; i++) { const [dataVal, fascia, riga, nome, entrata, uscita, mansione, annotazioni] = data[i]; const key = dataVal ? dataVal.toString().trim() : ''; if (!result[key]) continue; if (fascia && riga !== '' && result[key].fasce[fascia] && result[key].fasce[fascia][riga]) { result[key].fasce[fascia][riga] = { nome: nome||'', entrata: entrata||'', uscita: uscita||'', mansione: mansione||'' }; } if (annotazioni) result[key].annotazioni = annotazioni; } return {data: result}; } function handleSave(data) { const sh = getSheet(); const keys = Object.keys(data); const existing = sh.getDataRange().getValues(); const keyRows = {}; for (let i = 1; i < existing.length; i++) { const k = existing[i][0]?.toString().trim(); if (!keyRows[k]) keyRows[k] = []; keyRows[k].push(i + 1); } const rowsToDelete = []; keys.forEach(k => { if (keyRows[k]) rowsToDelete.push(...keyRows[k]); }); rowsToDelete.sort((a,b)=>b-a).forEach(r => sh.deleteRow(r)); const newRows = []; keys.forEach(key => { const dayData = data[key]; const annotazioni = dayData.annotazioni || ''; const fasce = dayData.fasce || {}; let firstRow = true; Object.entries(fasce).forEach(([fascia, rows]) => { rows.forEach((r, ri) => { newRows.push([ key, fascia, ri, r.nome||'', r.entrata||'', r.uscita||'', r.mansione||'', firstRow ? annotazioni : '' ]); firstRow = false; }); }); }); if (newRows.length > 0) { sh.getRange(sh.getLastRow()+1, 1, newRows.length, 8).setValues(newRows); } }
